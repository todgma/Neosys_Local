'use client';

import React, { useState } from 'react';
import { PropertyItem } from '@/types/users/UserPropertyItem';
import { ARTICLE_TYPE_LABELS, ArticleType, CAT3_LABELS, CA3Type, TRADE_TYPE_LABELS, tradeType } from '@/types/commons';

interface ReAdvertiseConfirmModalProps {
  isOpen: boolean;
  onClose: () => void;
  onConfirm: (deleteOriginal: boolean, removeBugiLink: boolean) => void;
  property: PropertyItem | null;
  bugiNo?: string;
}

// Helper function to get combined category label
const getCombinedCategoryLabel = (cat1: string, cat2?: string): string => {
  if (cat2 == null || cat2 == undefined || cat2 == '' || cat1 == cat2) {
    return ARTICLE_TYPE_LABELS[cat1 as ArticleType] || '-';
  }

  const cat1Label = ARTICLE_TYPE_LABELS[cat1 as ArticleType];
  const cat2Label = CAT3_LABELS[cat2 as CA3Type];

  if (cat1Label == cat2Label) {
    return cat1Label || '-';
  }

  return `${cat1Label} (${cat2Label})`;
};

export default function ReAdvertiseConfirmModal({
  isOpen,
  onClose,
  onConfirm,
  property,
  bugiNo
}: ReAdvertiseConfirmModalProps) {
  const [deleteOriginal, setDeleteOriginal] = useState(false);
  const [removeBugiLink, setRemoveBugiLink] = useState(false);

  if (!isOpen || !property) return null;

  // 부기사 연동 여부 확인 (BUGI_NO가 있고 '+'가 포함되지 않은 경우)
  const hasBugiLink = bugiNo && !bugiNo.includes('+');

  const handleConfirm = () => {
    onConfirm(deleteOriginal, removeBugiLink);
    setDeleteOriginal(false); // 리셋
    setRemoveBugiLink(false); // 리셋
  };

  const handleClose = () => {
    setDeleteOriginal(false); // 리셋
    setRemoveBugiLink(false); // 리셋
    onClose();
  };

  // 물건 정보 추출
  const { PT_NO, CAT2, CAT1, CAT0, TRD_TYP, PRIC1, PRIC2, ADDR1, ADDR2a, ADDR2b, ADDR3, ADDR4, BUILD_NAME, DONG_NAM, HO } = property;

  // 주소 조합 (PropertyTable과 동일한 로직)
  const mainAddress = ADDR2a
    ? `${ADDR1} ${ADDR2a} ${ADDR2b}`
    : `${ADDR1} ${ADDR2b}`;

  // 상세 주소 조합
  const detailAddressParts: string[] = [];
  if (BUILD_NAME) detailAddressParts.push(BUILD_NAME);

  if (CAT0 === 'SYCA001') {
    // 아파트인 경우
    if (DONG_NAM) detailAddressParts.push(`${DONG_NAM}동`);
    if (HO) detailAddressParts.push(`${HO}호`);
  } else {
    // 아파트가 아닌 경우
    if (ADDR3) detailAddressParts.push(ADDR3);
    if (ADDR4) detailAddressParts.push(ADDR4);
  }

  const detailAddress = detailAddressParts.join(' ');
  const address = detailAddress ? `${mainAddress} ${detailAddress}` : mainAddress;

  // 거래유형 라벨
  const tradeTypeLabel = TRD_TYP ? TRADE_TYPE_LABELS[TRD_TYP as tradeType] || TRD_TYP : '-';

  // 금액 포맷팅
  const formatPrice = (price?: number | null) => {
    if (!price) return '-';
    return `${price.toLocaleString()}만원`;
  };

  const price1Text = formatPrice(PRIC1);
  const price2Text = formatPrice(PRIC2);
  const priceText = TRD_TYP === 'SYTR001'
    ? `${price1Text}` // 매매
    : `${price1Text} / ${price2Text}`; // 전월세

  // 물건유형 (PropertyTable과 동일하게 CAT1, CAT2 순서)
  const categoryLabel = CAT1 ? getCombinedCategoryLabel(CAT1, CAT2) : '-';

  return (
    <div className="fixed inset-0 flex items-center justify-center z-50 p-4 bg-black/50 backdrop-blur-sm">
      <div className="bg-white rounded-2xl max-w-2xl w-full shadow-2xl border border-gray-100">
        {/* 헤더 */}
        <div className="border-b border-gray-100 p-6">
          <div className="flex items-center justify-between">
            <h2 className="text-2xl font-bold text-gray-900">재광고 확인</h2>
            <button
              onClick={handleClose}
              className="text-gray-400 hover:text-gray-600 transition-colors p-2 hover:bg-gray-100 rounded-lg"
            >
              <svg className="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
              </svg>
            </button>
          </div>
        </div>

        {/* 본문 */}
        <div className="p-6 space-y-6">
          {/* 안내 메시지 */}
          <div className="bg-blue-50 border border-blue-200 rounded-xl p-4">
            <p className="text-sm text-blue-800 font-semibold">
              아래 물건을 복사하여 재광고하시겠습니까?
            </p>
          </div>

          {/* 물건 요약 정보 */}
          <div className="bg-gray-50 rounded-xl p-5 space-y-3">
            <h3 className="text-lg font-bold text-gray-900 mb-4">재광고 대상 물건 정보</h3>

            <div className="grid grid-cols-2 gap-4">
              {/* 물건번호 */}
              <div className="flex items-center gap-3">
                <span className="text-sm font-semibold text-gray-600 w-24">물건번호</span>
                <span className="text-sm text-gray-900 font-medium">{PT_NO}</span>
              </div>

              {/* 물건유형 */}
              <div className="flex items-center gap-3">
                <span className="text-sm font-semibold text-gray-600 w-24">물건유형</span>
                <span className="text-sm text-gray-900 font-medium">{categoryLabel}</span>
              </div>

              {/* 거래유형 */}
              <div className="flex items-center gap-3">
                <span className="text-sm font-semibold text-gray-600 w-24">거래유형</span>
                <span className="text-sm text-gray-900 font-medium">{tradeTypeLabel}</span>
              </div>

              {/* 금액 */}
              <div className="flex items-center gap-3">
                <span className="text-sm font-semibold text-gray-600 w-24">금액</span>
                <span className="text-sm text-gray-900 font-medium">{priceText}</span>
              </div>

              {/* 소재지 */}
              <div className="col-span-2 flex items-center gap-3">
                <span className="text-sm font-semibold text-gray-600 w-24">소재지</span>
                <span className="text-sm text-gray-900 font-medium">{address || '-'}</span>
              </div>
            </div>
          </div>

          {/* 옵션 섹션 */}
          <div className="space-y-3">
            {/* 부기사 연동 옵션 - 부기사 연동이 있을 때만 표시 */}
            {hasBugiLink && (
              <div className="border border-orange-200 bg-orange-50 rounded-xl p-4">
                <div className="mb-3 pb-3 border-b border-orange-200">
                  <h4 className="text-sm font-bold text-orange-900 flex items-center gap-2">
                    <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                      <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13 16h-1v-4h-1m1-4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
                    </svg>
                    부기사 연동 안내
                  </h4>
                  <p className="text-xs text-orange-700 mt-2">
                    현재 부기사 플랫폼에 연동되어 있는 물건입니다. (부기사 물건번호: {bugiNo})
                  </p>
                </div>
                <label className="flex items-center gap-3 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={removeBugiLink}
                    onChange={(e) => setRemoveBugiLink(e.target.checked)}
                    className="w-5 h-5 text-orange-500 rounded focus:ring-orange-500 border-gray-300"
                  />
                  <div>
                    <span className="text-sm font-semibold text-gray-900">부기사 연동 해제 후 재광고</span>
                    <p className="text-xs text-gray-500 mt-1">부기사와의 연동을 해제하고 새 물건로 등록합니다.</p>
                    <p className="text-xs text-gray-500 mt-1">부기사에서 다올려물건 확인이 불가능합니다.</p>
                  </div>
                </label>
              </div>
            )}

            {/* 삭제 옵션 */}
            <div className="border border-gray-200 rounded-xl p-4">
              <label className="flex items-center gap-3 cursor-pointer">
                <input
                  type="checkbox"
                  checked={deleteOriginal}
                  onChange={(e) => setDeleteOriginal(e.target.checked)}
                  className="w-5 h-5 text-blue-500 rounded focus:ring-blue-500 border-gray-300"
                />
                <div>
                  <span className="text-sm font-semibold text-gray-900">기존 물건 삭제 후 새로 등록</span>
                  <p className="text-xs text-gray-500 mt-1">현재 물건을 완전히 삭제한 후 새 물건로 등록합니다.(물건번호가 변경됩니다.)</p>
                </div>
              </label>
            </div>
          </div>

          {/* 검증 정보 확인 안내 - 제일 아래 */}
          <div className="bg-red-50 border border-red-200 rounded-xl p-4">
            <p className="text-sm text-red-600 font-bold">
              재광고 진행 시 검증방식 선택 팝업이 열립니다. 검증 정보(매도자명, 소유자명, 전화번호 등)를 반드시 다시 확인해 주세요.
            </p>
          </div>
        </div>

        {/* 하단 버튼 */}
        <div className="border-t border-gray-100 p-6">
          <div className="flex gap-3">
            <button
              onClick={handleClose}
              className="flex-1 py-3 px-6 rounded-xl border-2 border-gray-300 text-gray-700 font-semibold hover:bg-gray-50 hover:border-gray-400 transition-all"
            >
              취소
            </button>
            <button
              onClick={handleConfirm}
              className="flex-1 py-3 px-6 rounded-xl bg-blue-600 hover:bg-blue-700 text-white font-semibold transition-all shadow-md hover:shadow-lg"
            >
              재광고 진행
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}
